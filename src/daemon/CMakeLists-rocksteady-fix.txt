# Rocksteady – CPU monitoring
add_executable(rocksteady
    rocksteady.c
    fugitoid_log.c
    krang.c
)

target_include_directories(rocksteady
    PRIVATE
        ${CMAKE_SOURCE_DIR}/src/core/include
        ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(rocksteady
    PRIVATE
        sensei_core
        crypto
)

install(TARGETS rocksteady
    RUNTIME DESTINATION bin
)
